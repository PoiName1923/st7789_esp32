# GIFs are optional: whichever of these exist in assets/gifs/ get embedded; the rest
# fall back to the built-in drawn face. Adding/removing a file re-runs CMake on the next
# build (directory is a configure dependency); if it doesn't, run `idf.py reconfigure`.
set(gif_names robot_idle robot_blink robot_happy robot_sleepy robot_surprised
              robot_look_left robot_look_right robot_weather_sunny robot_weather_rain)
get_filename_component(gif_dir "${COMPONENT_DIR}/../../assets/gifs" ABSOLUTE)

set(srcs "animation.c" "animation_name.c" "face.c" "gif_assets.c")
set(embed "")

if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    set(decl "")
    set(rows "")
    if(CONFIG_ROBOT_GIF_EMBEDDED)
        foreach(n ${gif_names})
            if(EXISTS "${gif_dir}/${n}.gif")
                list(APPEND embed "${gif_dir}/${n}.gif")
                string(APPEND decl "extern const uint8_t _binary_${n}_gif_start[], _binary_${n}_gif_end[];\n")
                string(APPEND rows "    { \"${n}\", _binary_${n}_gif_start, _binary_${n}_gif_end },\n")
            endif()
        endforeach()
        list(LENGTH embed n_embed)
        message(STATUS "animation: embedding ${n_embed} GIF(s) from ${gif_dir}")
    endif()
    set(gen "${CMAKE_CURRENT_BINARY_DIR}/gif_assets_table.c")
    file(CONFIGURE OUTPUT "${gen}" @ONLY CONTENT
"/* Generated by components/animation/CMakeLists.txt - do not edit */
#include \"gif_assets.h\"
@decl@
const gif_asset_t g_gif_assets[] = {
@rows@    { 0 }   /* sentinel: keeps the array non-empty */
};
const unsigned g_gif_asset_count = sizeof(g_gif_assets) / sizeof(g_gif_assets[0]) - 1;
")
    list(APPEND srcs "${gen}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${gif_dir}")
endif()

idf_component_register(SRCS ${srcs}
                       INCLUDE_DIRS "include"
                       EMBED_FILES ${embed}
                       REQUIRES lvgl__lvgl
                       PRIV_REQUIRES spiffs)

if(CONFIG_ROBOT_GIF_FROM_FS AND NOT CMAKE_BUILD_EARLY_EXPANSION)
    # Builds a SPIFFS image of assets/gifs and flashes it to the 'storage' partition.
    spiffs_create_partition_image(storage "${gif_dir}" FLASH_IN_PROJECT)
endif()
